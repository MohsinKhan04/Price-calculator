📜 Neon PostgreSQL DDL Schema (Run once in Neon SQL Editor) 

-- ========================================================
-- REFINECALC PRO - NEON POSTGRESQL SCHEMA
-- ========================================================

-- 1. Refineries Master Table
CREATE TABLE IF NOT EXISTS refineries (
    id SERIAL PRIMARY KEY,
    code VARCHAR(30) UNIQUE NOT NULL,
    name VARCHAR(150) NOT NULL,
    city VARCHAR(100) NOT NULL,
    state VARCHAR(100) NOT NULL,
    latitude DECIMAL(9, 6) NOT NULL,
    longitude DECIMAL(9, 6) NOT NULL,
    loading_capacity VARCHAR(100),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2. Transporters Master Table
CREATE TABLE IF NOT EXISTS transporters (
    id SERIAL PRIMARY KEY,
    name VARCHAR(150) NOT NULL,
    base_city VARCHAR(100) NOT NULL,
    rate_per_mt_per_km DECIMAL(6, 2) NOT NULL,
    min_freight DECIMAL(10, 2) DEFAULT 0,
    contact_phone VARCHAR(50),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 3. 15-Day Revision Price List Table
CREATE TABLE IF NOT EXISTS price_lists (
    id SERIAL PRIMARY KEY,
    cycle_start DATE NOT NULL,
    cycle_end DATE NOT NULL,
    product_name VARCHAR(100) NOT NULL,
    refinery_code VARCHAR(30) REFERENCES refineries(code) ON DELETE CASCADE,
    basic_price DECIMAL(10, 2) NOT NULL,
    discount DECIMAL(10, 2) DEFAULT 0,
    net_price DECIMAL(10, 2) GENERATED ALWAYS AS (basic_price - discount) STORED,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 4. Customer Quotes Audit Log
CREATE TABLE IF NOT EXISTS quote_logs (
    id SERIAL PRIMARY KEY,
    product_name VARCHAR(100) NOT NULL,
    destination_city VARCHAR(100) NOT NULL,
    quantity_mt DECIMAL(10, 2) NOT NULL,
    chosen_refinery_code VARCHAR(30),
    chosen_transporter_id INT,
    distance_km DECIMAL(8, 2),
    basic_price DECIMAL(10, 2),
    discount DECIMAL(10, 2),
    net_price DECIMAL(10, 2),
    transport_cost DECIMAL(10, 2),
    margin DECIMAL(10, 2),
    customer_price DECIMAL(10, 2),
    total_order_val DECIMAL(12, 2),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

⚡ Lightweight Render Backend Server (server.js for Node.js) 

// server.js - Node.js Express backend to connect to Neon PostgreSQL on Render
const express = require('express');
const { Pool } = require('pg');
const cors = require('cors');

const app = express();
app.use(cors());
app.use(express.json());
app.use(express.static('.')); // Serves index.html directly!

// Neon Database Connection String (set in Render Environment Variables)
const pool = new Pool({
  connectionString: process.env.DATABASE_URL,
  ssl: { rejectUnauthorized: false }
});

// GET active price list & master data
app.get('/api/masters', async (req, res) => {
  try {
    const refineries = await pool.query('SELECT * FROM refineries WHERE is_active = true');
    const transporters = await pool.query('SELECT * FROM transporters WHERE is_active = true');
    const prices = await pool.query('SELECT * FROM price_lists ORDER BY cycle_start DESC');
    res.json({ refineries: refineries.rows, transporters: transporters.rows, prices: prices.rows });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// POST new 15-day price list batch
app.post('/api/pricelist/batch', async (req, res) => {
  const { items } = req.body;
  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    for (const item of items) {
      await client.query(
        'INSERT INTO price_lists (cycle_start, cycle_end, product_name, refinery_code, basic_price, discount) VALUES ($1, $2, $3, $4, $5, $6)',
        [item.cycle_start, item.cycle_end, item.product_name, item.refinery_code, item.basic_price, item.discount]
      );
    }
    await client.query('COMMIT');
    res.json({ success: true, count: items.length });
  } catch (err) {
    await client.query('ROLLBACK');
    res.status(500).json({ error: err.message });
  } finally {
    client.release();
  }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`RefineCalc running on port ${PORT}`));
